4) write a code to find missing number in Integer List
    Test cases :
       a) 1,2,3,4,5,7,8,9,10

       c)1,3,5,9,11,13,15,17
import java.util.*;
public class MissingNumberFinder {
    public static Integer findMissingNumber(List<Integer> list) {
        if (list == null || list.size() < 2) {
            return null;
        }
        // Find the common difference (minimum gap)
        int diff = Integer.MAX_VALUE;
        for (int i = 1; i < list.size(); i++) {
            diff = Math.min(diff, list.get(i) - list.get(i - 1));
        }

        // Find the missing element
        for (int i = 1; i < list.size(); i++) {
            if (list.get(i) - list.get(i - 1) != diff) {
                return list.get(i - 1) + diff;
            }
        }
        return null;
    }

    public static void main(String[] args) {
        List<Integer> a = Arrays.asList(1,2,3,4,5,7,8,9,10);
        System.out.println("a -> " + findMissingNumber(a));
    }
}

import java.util.Arrays;
public class QuickSortExample {
    public static void quickSort(int[] arr, int low, int high) {
        if (low < high) {
            int pivot = partition(arr, low, high);
            quickSort(arr, low, pivot - 1);
            quickSort(arr, pivot + 1, high);
        }
    }
    private static int partition(int[] arr, int low, int high) {
        int pivot = arr[high];
        int i = low - 1;
        for (int j = low; j < high; j++) {
            if (arr[j] <= pivot) {
                i++;
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
        int temp = arr[i + 1];
        arr[i + 1] = arr[high];
        arr[high] = temp;
        return i + 1;
    }
    public static void main(String[] args) {
        int[] arr = {2, 4, 0, 1, 56, 23, 4};
        quickSort(arr, 0, arr.length - 1);
        System.out.println(Arrays.toString(arr));
    }
}

5. Custom Collector to Partition Primes and Non-Primes

public class PrimePartition {
    public static void main(String[] args) {
        List<Integer> numbers = Arrays.asList(2, 3, 4, 5, 6, 7, 8, 9);
        
        Map<Boolean, List<Integer>> partitioned = numbers.stream()
            .collect(Collectors.partitioningBy(PrimePartition::isPrime));
        
        System.out.println("Primes: " + partitioned.get(true));
        System.out.println("Non-Primes: " + partitioned.get(false));
    }

    private static boolean isPrime(int number) {
        if (number <= 1) return false;
        return IntStream.range(2, number).noneMatch(i -> number % i == 0);
    }
}



6. Find the Highest Transaction in Each Month

class Transaction {
    int amount;
    String month;

    public Transaction(int amount, String month) {
        this.amount = amount;
        this.month = month;
    }
}

public class HighestTransaction {
    public static void main(String[] args) {
        List<Transaction> transactions = Arrays.asList(
            new Transaction(100, "January"),
            new Transaction(200, "January"),
            new Transaction(150, "February"),
            new Transaction(300, "February")
        );
        
        Map<String, Optional<Transaction>> highestPerMonth = transactions.stream()
            .collect(Collectors.groupingBy(
                tx -> tx.month, 
                Collectors.maxBy(Comparator.comparingInt(tx -> tx.amount))
            ));
        
        highestPerMonth.forEach((month, transaction) -> 
            System.out.println(month + ": " + transaction.get().amount)
        );
    }
}

public class Pagination {

        List<Integer> numbers = IntStream.rangeClosed(1, 100).boxed().collect(Collectors.toList());
        
        int pageSize = 5;
        int page = 3; // Fetch the 2nd page (6th to 10th elements)

                                             .skip((page - 1) * pageSize)
                                             .limit(pageSize)
                                             .collect(Collectors.toList());
        
        System.out.println("Page " + page + ": " + paginatedList);
    }

12.Write a program to find the intersection (common elements) of two lists using Streams.

public static void main(String[] args) {
        List<Integer> list1 = Arrays.asList(1, 2, 3, 4, 5);
        List<Integer> list2 = Arrays.asList(4, 5, 6, 7, 8);

        // Convert list2 to a Set for efficient lookups
        Set<Integer> set2 = new HashSet<>(list2); // More Efficient Approach (Using Set):

        // Finding the intersection using Streams
        List<Integer> intersection = list1.stream()
                                          .filter(set2::contains)  // Check if list1 element is in set2
                                          .collect(Collectors.toList());

        // Printing the intersection
        System.out.println("Intersection: " + intersection);  // Output: [4, 5]
    }
